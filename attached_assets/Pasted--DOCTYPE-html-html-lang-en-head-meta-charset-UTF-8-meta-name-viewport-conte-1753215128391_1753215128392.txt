  <script type="module">
    import { initializeApp } from 'https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js';
    import { getFirestore, collection, getDocs, query, orderBy } from 'https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js';
    import { getStorage, ref, getDownloadURL } from 'https://www.gstatic.com/firebasejs/10.12.2/firebase-storage.js';

    const firebaseConfig = {
      apiKey: "AIzaSyBNAiBbjBl7SpqwL0NiJmx3BVuRrd1FVBA",
      authDomain: "finafeels-39852.firebaseapp.com",
      projectId: "finafeels-39852",
      storageBucket: "finafeels-39852.appspot.com",
      messagingSenderId: "962083690885",
      appId: "1:962083690885:web:115469c62f982fa24fccec",
      measurementId: "G-C60L4MRLYR"
    };

    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);
    const storage = getStorage(app);

    const feed = document.getElementById('collab-feed');
    const collabRef = collection(db, 'collaborations');
    const q = query(collabRef, orderBy('date', 'desc'));

    const renderCollab = async (data) => {
      const section = document.createElement('section');
      section.className = 'collab-section';

      section.innerHTML = `
        <h2 class="collab-title">${data.title}</h2>
        <p class="collab-desc">${data.description}</p>
        <div class="collab-media"></div>
      `;

      const mediaContainer = section.querySelector('.collab-media');

      for (const path of data.images || []) {
        const url = await getDownloadURL(ref(storage, path));
        const img = document.createElement('img');
        img.src = url;
        mediaContainer.appendChild(img);
      }

      if (data.video) {
        const videoURL = await getDownloadURL(ref(storage, data.video));
        const video = document.createElement('video');
        video.controls = true;
        video.className = 'no-print';
        video.src = videoURL;
        mediaContainer.appendChild(video);
      }

      feed.appendChild(section);
    };

    const loadCollabs = async () => {
      const snapshot = await getDocs(q);
      snapshot.forEach(doc => renderCollab(doc.data()));
    };

    loadCollabs();
  </script>
